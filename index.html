            <input type="number" id="beans" placeholder="如：1200">

    <script>

        // ==========================================

        // 核心配置区：后续如果位置微调，修改这里的百分比数字即可

        // ==========================================

        const layout = {

            // x 都为 0.5 (即50%，表示水平居中)

            branch:  { x: 0.5, y: 0.66, size: 0.045 }, // 支行名称 字号占宽度的4.5%

            manager: { x: 0.5, y: 0.715, size: 0.03 }, // 客户经理

            product: { x: 0.5, y: 0.755, size: 0.03 }, // 产品

            amount:  { x: 0.5, y: 0.795, size: 0.035 }, // 金额

            beans:   { x: 0.5, y: 0.835, size: 0.035 }, // 金豆

            date:    { x: 0.5, y: 0.89, size: 0.025 }  // 日期

        };

        const textColor = "#8B0000"; // 深红色

        const fontFamily = "'Microsoft YaHei', 'PingFang SC', sans-serif";


        // 加载背景图（注意：请将你的背景图命名为 bg.jpg 并和此 html 放在同一目录）

        const bgImage = new Image();

        bgImage.src = 'bg.jpg'; 

        let isImageLoaded = false;


        bgImage.onload = function() {

            isImageLoaded = true;

            // 页面加载后可以先在控制台打印一下背景图尺寸，确认是否正常

            console.log("背景图加载成功，尺寸:", bgImage.width, bgImage.height);

        };


        function generatePoster() {

            if (!isImageLoaded) {

                alert("背景图尚未加载完成或路径错误，请确保 bg.jpg 与本网页在同一文件夹！");

                return;

            }


            const branch = document.getElementById('branch').value.trim();

            const manager = document.getElementById('manager').value.trim();

            const product = document.getElementById('product').value.trim();

            const amount = document.getElementById('amount').value.trim();

            const beans = document.getElementById('beans').value.trim();


            if (!branch || !manager || !product || !amount || !beans) {

                alert("请将信息填写完整！");

                return;

            }


            // 获取今天的日期

            const today = new Date();

            const dateStr = `${today.getFullYear()}年${today.getMonth() + 1}月${today.getDate()}日`;


            const canvas = document.getElementById('posterCanvas');

            const ctx = canvas.getContext('2d');

            

            // 设置画布大小与背景图一致（保持高清）

            canvas.width = bgImage.width;

            canvas.height = bgImage.height;


            // 1. 绘制背景图

            ctx.drawImage(bgImage, 0, 0, canvas.width, canvas.height);


            // 2. 绘制文字

            ctx.textAlign = "center";

            ctx.textBaseline = "middle";

            ctx.fillStyle = textColor;


            // 辅助绘制函数

            const drawText = (text, config) => {

                const fontSize = canvas.width * config.size;

                ctx.font = `bold ${fontSize}px ${fontFamily}`;

                const x = canvas.width * config.x;

                const y = canvas.height * config.y;

                ctx.fillText(text, x, y);

            };


            // 渲染 6 项信息（按你的要求加入了后缀）

            drawText(branch, layout.branch);

            drawText(`客户经理 ${manager} 成功营销`, layout.manager);

            drawText(`产品：${product}`, layout.product);

            drawText(`销售金额：${amount} 万元`, layout.amount);

            drawText(`喜获金豆：${beans} 颗`, layout.beans);

            drawText(dateStr, layout.date);


            // 显示 Canvas 和下载按钮

            canvas.style.display = "block";

            document.getElementById('downloadBtn').style.display = "block";

        }


        function downloadImage() {

            const canvas = document.getElementById('posterCanvas');

            const link = document.createElement('a');

            link.download = '销售喜报.png';

            link.href = canvas.toDataURL('image/png');

            link.click();

        }
